```sql
DELIMITER ;;

DROP PROCEDURE IF EXISTS up_sync_pack_quantities;;
CREATE PROCEDURE up_sync_pack_quantities ()
BEGIN
    DECLARE EXIT HANDLER FOR SQLEXCEPTION
    BEGIN
        -- Cleanup on error
        DROP TEMPORARY TABLE IF EXISTS t_pack_deltas;
        RESIGNAL;
    END;

    -- use a temporary table because CTE's are ephemeral and cross-talk between
    -- connections will be a major problem
    DROP TEMPORARY TABLE IF EXISTS t_pack_deltas;
    CREATE TEMPORARY TABLE t_pack_deltas AS
    WITH product_stock AS (
        SELECT
            p.id_product_pack,
            p.id_product_item,
            p.id_product_attribute_item,
            CAST(p.quantity AS REAL) AS pack_quantity,
            CAST(SUM(a.quantity) AS REAL) AS item_quantity
        FROM ps_pack p
        LEFT JOIN ps_swt_customer_configurator c ON p.id_product_pack = c.id_pack_product
        INNER JOIN ps_stock_available a ON p.id_product_item = a.id_product AND p.id_product_attribute_item = a.id_product_attribute
        WHERE c.id_pack_product IS NULL
        GROUP BY
            p.id_product_pack,
            p.id_product_item,
            p.id_product_attribute_item,
            p.quantity
    ),
    pack_quantities AS (
        SELECT
            id_product_pack,
            FLOOR(MIN(item_quantity / pack_quantity)) AS min_pack_quantity
        FROM product_stock
        GROUP BY
            id_product_pack
    )
    -- actually insert into the temp table here
    SELECT
        id_product_pack,
        min_pack_quantity,
        s.quantity AS existing_quantity,
        min_pack_quantity - s.quantity AS delta
    FROM pack_quantities q
    INNER JOIN ps_stock_available s ON q.id_product_pack = s.id_product AND s.id_product_attribute = 0
    WHERE min_pack_quantity - s.quantity <> 0;

    -- update the actual stock numbers if there are any values in the temp table
    UPDATE ps_stock_available s
    INNER JOIN t_pack_deltas d ON s.id_product = d.id_product_pack AND s.id_product_attribute = 0
    SET s.quantity = d.min_pack_quantity;

    -- insert the relevant information into ps_stock_mvt
    INSERT INTO ps_stock_mvt
        (id_stock, id_stock_mvt_reason, id_employee, employee_lastname, employee_firstname, physical_quantity, date_add, sign)
    SELECT
        s.id_stock_available,
        IF(d.delta > 0, 11, 12),
        1,
        'Brickman',
        'Kris',
        ABS(d.delta),
        CURRENT_TIMESTAMP(),
        IF(d.delta > 0, 1, -1)
    FROM t_pack_deltas d
    INNER JOIN ps_stock_available s ON d.id_product_pack = s.id_product AND s.id_product_attribute = 0;

END;;
```

```php
    public static function getQuantityAvailableByProduct($id_product = null, $id_product_attribute = null, $id_shop = null)
    {
        // if null, it's a product without attributes
        if ($id_product_attribute === null) {
            $id_product_attribute = 0;
        }

        $key = 'StockAvailable::getQuantityAvailableByProduct_' . (int) $id_product . '-' . (int) $id_product_attribute . '-' . (int) $id_shop;
        if (!Cache::isStored($key)) {
            $query = new DbQuery();
            $query->select('SUM(quantity)');
            $query->from('stock_available');

            // if null, it's a product without attributes
            if ($id_product !== null) {
                $query->where('id_product = ' . (int) $id_product);
            }

            $query->where('id_product_attribute = ' . (int) $id_product_attribute);
            $query = StockAvailable::addSqlShopRestriction($query, $id_shop);
            $result = (int) Db::getInstance(_PS_USE_SQL_SLAVE_)->getValue($query);
            Cache::store($key, $result);

            return $result;
        }

        return Cache::retrieve($key);
    }
```

```php
public function hookActionUpdateQuantity($params)
{
    if (!Configuration::get("BUNDLESSTOCKS_AUTO")) {
        return;
    }

    $id_shop = isset($params['id_shop']) ? (int)$params['id_shop'] : (int)$this->context->shop->id;
    $id_product = (int)$params['id_product'];
    $id_product_attribute = (int)$params['id_product_attribute'];

    if (isset(self::$qty_update_hook_called[$id_shop][$id_product][$id_product_attribute])) {
        return;
    }

    self::$qty_update_hook_called[$id_shop][$id_product][$id_product_attribute] = true;
    self::log("hookActionUpdateQuantity for Product #".$id_product.' (attribute #'.$id_product_attribute.')');
    self::bulkUpdatePacks();
}

// this is called by the cron script, although it's mostly unnecessary at this point
public function updateAllPacks($id_product = null, $id_shop = null)
{
    self::bulkUpdatePacks();
}

private static function bulkUpdatePacks() {
    self::log("bulkUpdatePacks: handling all pack counts");

    $updates = Db::getInstance()->executeS("CALL up_sync_pack_quantities();");
    foreach($updates as $update) {
        $key = 'StockAvailable::getQuantityAvailableByProduct_' . (int) $update['id_product_pack'] . '_0_1';
        $value = (int) $update['quantity'];

        Cache::set($key, $value);
    }
}
```

so anything that calls `hook("actionUpdateQuantity");` with any params will update the whole pack inventory (ignoring the params).  the `updateAllPacks` method was/is called by that cron job.  both of them funnel into the private `bulkUpdatePacks()` method which is where the sproc is actually called

install:
```sql
CREATE TABLE ps_search_aggregate
(
    id INT AUTO_INCREMENT NOT NULL PRIMARY KEY,
    id_product INT NOT NULL,
    id_lang INT NOT NULL,
    id_shop INT NOT NULL,
    id_image INT NULL,
    product_name VARCHAR(255) NOT NULL,
    product_description VARCHAR(255) NOT NULL,
    category_name VARCHAR(255) NULL,
    manufacturer_name VARCHAR(255) NULL,
    price DECIMAL(8,2) NOT NULL DEFAULT 0.0,
    quantity INT NOT NULL,
    link_rewrite VARCHAR(255) NULL,
    weight DECIMAL(8,2) NOT NULL DEFAULT 0.0,
    combined TEXT NULL,
    FULLTEXT combined_idx (combined)
) ENGINE=InnoDB;
```

run on "change" event:
```sql
TRUNCATE TABLE ps_search_aggregate;
INSERT INTO ps_search_aggregate (
    id_product,
    id_lang,
    id_shop,
    id_image,
    product_name,
    product_description,
    category_name,
    manufacturer_name,
    price,
    quantity,
    link_rewrite,
    weight,
    combined
) SELECT
    p.id_product,
    pl.id_lang,
    ps.id_shop,
    i.id_image,
    pl.name AS product_name,
    IF (
        LENGTH(pl.description_short) > 253,
        CONCAT(LEFT(pl.description_short, 250), '...'),
        pl.description_short
    ) AS product_description,
    cl.name AS category_name,
    m.name AS manufacturer_name,
    p.price,
    sa.quantity,
    pl.link_rewrite,
    COALESCE(w.calculated_weight, 0) + (COALESCE(w.manual_boost, 0) * 100) AS weight,
    CONCAT(
        COALESCE(pl.name, ''), ' ',
        COALESCE(pl.description_short, ''), ' ',
        COALESCE(cl.name, ''), ' ',
        COALESCE(m.name, '')
    ) AS combined
FROM `ps_product` p
LEFT JOIN `ps_product_lang` pl ON (p.id_product = pl.id_product)
LEFT JOIN `ps_category_lang` cl ON (p.id_category_default = cl.id_category)
LEFT JOIN `ps_manufacturer` m ON (p.id_manufacturer = m.id_manufacturer)
LEFT JOIN `ps_stock_available` sa ON (p.id_product = sa.id_product AND sa.id_product_attribute = 0)
LEFT JOIN `ps_image` i ON (p.id_product = i.id_product AND i.cover = 1)
LEFT JOIN `ps_product_shop` ps ON (p.id_product = ps.id_product)
LEFT JOIN `ps_category` c ON (p.id_category_default = c.id_category)
LEFT JOIN `ps_prestosearch_weights` w ON (p.id_product = w.product_id)
WHERE p.active = 1 AND ps.active = 1 AND c.active = 1;
```

"new" search:
```sql
SELECT *, MATCH(combined) AGAINST ('Bondtech' IN BOOLEAN MODE) AS relevance_score
FROM ps_search_aggregate
WHERE
  MATCH(combined) AGAINST ('Bondtech' IN BOOLEAN MODE)
ORDER BY
  (weight + relevance_score) DESC,
  price DESC,
  product_name ASC
  LIMIT 25;
```

although the search part is missing the additional parts you had for "exact match = 10000 relevance" and the mroe rigid filters like price.  it seems like the ordering should be weight * relevance, not weight + relevance, but it's easy to play with

[7/24/2025 10:32 AM] brett4091
At the most basic, ProductBundles are just virtual containers comprised of Products.

Example:

- ProductBundle X contains 2x Product A and 4x Product B.
- Product A currently has 10 stock, Product B currently has 12 stock.
- ProductBundle X can be bought 3 times

What *could* be a little tricky with this implementation is that ProductBundles "inventory" should be recalculated based on the individual Products inventory, triggered by events such as Orders.


[7/24/2025 10:50 AM] brett4091
I may be misunderstanding your intention for ProductBundles, but for now I am assuming these are pack equivalents. With that said, these are my thoughts, but you and Kris may disagree

- Inventory always represents physical products. Digital products like manuals, gift cards, etc. and Services have no such restriction. If they do, the allocations should be tracked separately from Inventory.

- Pricing is at the product level. Inventory is strictly the quantity of the physical item available.

- Products and ProductBundles can both be added to carts. ProductBundles are a collection of Products, grouped for simplicity and potentially "bought together" discounts.
Hardware packs like the [Sherpa Heavy Extruder - Hardware Pack - RIGDA GearSet](https://kb-3d.com/store/extruders-hotends/1389-sherpa-heavy-extruder-hardware-pack-rigda-gearset-1723833309605.html) provide a good example. It contains a bunch of individual Products, customers could add all of these to the cart separately, but the ProductBundles save them the trouble.

- A new Product does not inherently create a ProductBundle. The ProductBundle is an optional way to group Products, but 99% of the time Products will be standalone

1) When a cart is sold, it will broadcast several different messages, but one of them is a "InventoryQuantityReduced" message
2) This message is picked up by the Inventory module, which subtracts the quantity, and notifies the Products about this change
3) Any Product for this Inventory will analyse the new quantity, and conditionally make themselves Available
4) Any Product that is now Available then produces a "ProductQuantityUpdated" event

5) Everyone else's carts immediately pick up this event, and compare the quantity in their cart with the quantity that was just announced in that event
6) Any carts with less quantity than available are unaffected, and proceed as normal
7) Any carts with more quantity than is remaining will go into the Backorder workflow
8) Backordered carts will check the Backorder rules for all their Products
9) If all Products in the cart allow backordering, then the Customer is shown a notification on the site that a Product in their Cart is out of stock but can be backordered

10) If the Customer decides to backorder their Cart, the checkout process goes through as normal, but the Cart is held in a "Pending" state
11) The "CartStatusPending" event can be used to notify you that someone backordered something
12) When the Inventory is restocked, the backoffice portal generates an "InventoryQuantityIncreased" message
13) This message is picked up by the Inventory module, which changes itself from Backordered to Available
14) The Product module produces a "ProductQuantityUpdated" event
15) This message propagates down to all the current Carts, and if any of them are backordered for this Product, then they are woken back up

16) All Carts in the "Pending" state immediately try and complete their checkout again, restarting back at Step #1


[8/4/2025 1:50 PM] instazx2
this gives us:

* No negative inventory quantity ever
* Inventory that can backorder itself if the quantity hits a threshold
* Products that have no quantity, like the acrylic panels
* Prioritizing backordered carts when stock becomes available
* Allows/Disallows backordering at an Inventory, Product, and Customer level
* Carts keep a local copy of the available Product quantities, so there's less DB overhead during high traffic
* Allows for carts to "sort" themselves when stock is getting low
* Allows a backordered Cart to be later cancelled without impacting stock quantity

[8/7/2025 8:53 AM] brett4091
```
Current e-commerce variant systems are often rigid and limited - typically handling simple attributes like size, color, and material with basic combinations. Here's what I'd propose for a more sophisticated approach:

Core Architecture Improvements

Hierarchical Variant Structure: Instead of flat attribute lists, use a tree structure where variants can have parent-child relationships. For example, "Material" could branch into "Cotton" which then branches into "Organic Cotton" and "Regular Cotton."
Dynamic Attribute System: Allow merchants to define custom attributes with validation rules, dependencies, and display logic. Attributes could have types (text, numeric, boolean, enum, range) with appropriate UI controls.

Conditional Logic Engine: Implement rules that show/hide certain combinations based on other selections. If someone picks "Leather" as material, automatically hide color options that don't make sense for leather.

Advanced Features

Multi-dimensional Variants: Support for complex products like configurable software, custom manufacturing, or services where variants aren't just physical attributes but could include duration, capacity, or feature sets.

Inventory Forecasting Integration: The variant system should predict demand for combinations and suggest which variants to stock based on historical data and current trends.
Visual Variant Builder: A drag-and-drop interface where merchants can visually map out variant relationships, set up rules, and preview how customers will experience the selection process.
Smart Defaults and Recommendations: Use ML to suggest the most popular or relevant variant combinations for each customer based on their browsing history, location, or similar customer behavior.

Technical Implementation

Graph Database Backend: Store variants and their relationships in a graph structure rather than traditional relational tables, allowing for more flexible querying and relationship mapping.
Real-time Validation: As customers select options, instantly validate availability, pricing changes, and compatibility without page reloads.
Headless API Design: Separate the variant logic from presentation, allowing the same system to power web, mobile, AR try-on experiences, and voice commerce.

The key insight is moving from "products with variants" to "configurable product systems" - treating each product as a small decision tree that can adapt to different business models and customer needs.
```
